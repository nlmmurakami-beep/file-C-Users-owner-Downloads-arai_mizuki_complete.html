let selectedCategory = '';
function filterArticles() {
const keyword = document.getElementById('search').value.trim().toLowerCase();
const articles = document.querySelectorAll('#article-list .article');
let visibleCount = 0;
articles.forEach((article) => {
const title = article.querySelector('h3').textContent.toLowerCase();
const desc = article.querySelector('p').textContent.toLowerCase();
const cat = article.dataset.cat || '';
const matchKeyword =
!keyword ||
title.includes(keyword) ||
desc.includes(keyword) ||
cat.toLowerCase().includes(keyword);
const matchCat = !selectedCategory || cat === selectedCategory;
if (matchKeyword && matchCat) {
article.style.display = '';
visibleCount++;
} else {
article.style.display = 'none';
}
});
document.getElementById('count').textContent = `検索結果：${visibleCount}件`;
}
function setCategory(cat) {
selectedCategory = cat;
filterArticles();
}
function updateRanking() {
const rankingList = document.getElementById('ranking-list');
const articles = Array.from(document.querySelectorAll('#article-list .article'));
const ranked = articles
.map((article) => {
const title = article.querySelector('h3').textContent;
const countEl = article.querySelector('.like-count');
const likes = Number(countEl ? countEl.textContent : 0) || 0;
return { title, likes };
})
.sort((a, b) => b.likes - a.likes);
rankingList.innerHTML = ranked
.map((item, idx) => `
<li class="ranking-item">
<div class="rank-left">
<span class="rank-num">${idx + 1}</span>
<span>${item.title}</span>
</div>
<span class="rank-score">♥ ${item.likes}</span>
</li>
`)
.join('');
}
document.addEventListener('DOMContentLoaded', () => {
document.querySelectorAll('.like-btn').forEach((btn) => {
const iconEl = btn.querySelector('.like-icon');
const textEl = btn.querySelector('.like-text');
const countEl = btn.querySelector('.like-count');
let count = Number(btn.dataset.count) || 0;
let isLiked = btn.classList.contains('is-liked');
const render = () => {
btn.classList.toggle('is-liked', isLiked);
btn.setAttribute('aria-pressed', String(isLiked));
if (iconEl) iconEl.textContent = isLiked ? '♥' : '♡';
if (textEl) textEl.textContent = isLiked ? 'いいね済み' : 'いいね';
if (countEl) countEl.textContent = String(count);
};
render();
btn.addEventListener('click', () => {
isLiked = !isLiked;
count = isLiked ? count + 1 : Math.max(0, count - 1);
render();
if (isLiked && iconEl) {
iconEl.style.animation = 'none';
void iconEl.offsetWidth;
iconEl.style.animation = '';
}
updateRanking();
});
});
updateRanking();

if (!heartsContainer) return;

heart.style.left = `${Math.random() * 100}%`;
heart.style.fontSize = `${16 + Math.random() * 14}px`;
heartsContainer.appendChild(heart);
setTimeout(() => heart.remove(), 8000);
}, 1400);
});
